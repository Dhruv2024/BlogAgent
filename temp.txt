from langgraph.pregel import RetryPolicy

def _is_retryable_429(exc: Exception) -> bool:
    status_code = getattr(exc, "status_code", None) or getattr(exc, "status", None)
    message = str(exc).lower()
    return status_code == 429 or "429" in message or "rate limit" in message

g.add_node(
    "router", router_node,
    retry=RetryPolicy(max_attempts=4, initial_interval=5.0, backoff_factor=2.0, retry_on=_is_retryable_429)
)
g.add_node(
    "research", research_node,
    retry=RetryPolicy(max_attempts=4, initial_interval=5.0, backoff_factor=2.0, retry_on=_is_retryable_429)
)
g.add_node(
    "orchestrator", orchestrator_node,
    retry=RetryPolicy(max_attempts=4, initial_interval=5.0, backoff_factor=2.0, retry_on=_is_retryable_429)
)
g.add_node(
    "worker", worker_node,
    retry=RetryPolicy(max_attempts=5, initial_interval=10.0, backoff_factor=2.0, retry_on=_is_retryable_429)
)
